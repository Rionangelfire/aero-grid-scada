} if (!db.objectStoreNames.contains('exports')) { db.createObjectStore('exports', { keyPath: 'id', autoIncrement: true }); } }; req.onsuccess = (e) => { dbInstance = e.target.result; resolve(dbInstance); }; req.onerror = (e) => reject(e.target.error); }); } function dbPut(storeName, record) { if (!dbInstance) return Promise.resolve(); return new Promise((resolve, reject) => { const tx = dbInstance.transaction(storeName, 'readwrite'); const store = tx.objectStore(storeName); const req = store.put(record); req.onsuccess = () => resolve(req.result); req.onerror = () => reject(req.error); }); } function dbGetAll(storeName) { if (!dbInstance) return Promise.resolve([]); req.onerror = () => reject(req.error); }); } function dbClearAll() { if (!dbInstance) return Promise.resolve(); return new Promise((resolve) => { const tx = dbInstance.transaction(['missions', 'incidents', 'exports'], 'readwrite'); try { await dbPut('missions', record); if (manual) { addLog(`Mission ${m.id} committed to Digital Twin IndexedDB Archive.`, 'success'); } } catch (err) { console.error("Failed to commit mission archive:", err);
